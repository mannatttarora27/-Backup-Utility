UNIX Backup Utility Script

#!/bin/bash
if [ -z "$1" ]; then
    echo "Usage: $0 "
    exit 1
fi
SOURCE=$1
BACKUP_DIR="/backup"
if [ ! -d "$BACKUP_DIR" ]; then
    sudo mkdir -p "$BACKUP_DIR"
    sudo chmod 755 "$BACKUP_DIR"
fi
DATE=$(date +'%d-%m-%Y')
NAME=$(basename "$SOURCE")
DEST="$BACKUP_DIR/${NAME}-${DATE}"
if [ -d "$SOURCE" ]; then
    cp -r "$SOURCE" "$DEST"
elif [ -f "$SOURCE" ]; then
    mkdir -p "$DEST"
    cp "$SOURCE" "$DEST/"
else
    echo "Error: $SOURCE does not exist."
    exit 1
fi
echo "✅ Backup of '$SOURCE' created successfully at: $DEST"
